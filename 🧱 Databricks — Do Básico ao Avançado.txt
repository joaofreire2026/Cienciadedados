## 🟡 NÍVEL INTERMEDIÁRIO

### 5. Arquitetura Medallion (Lakehouse)

```python
# ══════════════════════════════════════════════════
#  BRONZE → SILVER → GOLD
# ══════════════════════════════════════════════════

# ── BRONZE (Raw / Ingestão) ───────────────────────────────
"""
BRONZE = Dados brutos, como chegam da fonte
  • SEM transformações de negócio
  • Preservar dado original (auditoria)
  • Adicionar metadados de ingestão
  • Formato: Delta (preferencialmente)
"""

from datetime import datetime

def ingerir_bronze(caminho_origem: str, tabela_destino: str, formato: str = "csv"):
    df_raw = spark.read.format(formato) \
        .option("header", "true") \
        .load(caminho_origem)
    
    # Metadados de ingestão
    df_bronze = df_raw \
        .withColumn("_ingestion_timestamp", F.current_timestamp()) \
        .withColumn("_source_file", F.input_file_name()) \
        .withColumn("_ingestion_date", F.current_date())
    
    df_bronze.write \
        .format("delta") \
        .mode("append") \
        .option("mergeSchema", "true") \
        .saveAsTable(f"bronze.{tabela_destino}")
    
    print(f"✅ Bronze: {df_bronze.count()} registros ingeridos → {tabela_destino}")
    return df_bronze

# ── SILVER (Cleaned / Conformed) ─────────────────────────
"""
SILVER = Dados limpos e validados
  • Remover duplicatas
  • Corrigir tipos de dados
  • Padronizar valores
  • Aplicar regras de qualidade
  • Unir fontes relacionadas
"""

def processar_silver(tabela_bronze: str, tabela_silver: str):
    df = spark.read.table(f"bronze.{tabela_bronze}")
    
    df_silver = df \
        .dropDuplicates(["id", "data_evento"]) \
        .filter(F.col("id").isNotNull()) \
        .withColumn("valor", F.col("valor").cast("double")) \
        .withColumn("data_evento", F.to_date(F.col("data_str"), "dd/MM/yyyy")) \
        .withColumn("nome", F.trim(F.upper(F.col("nome")))) \
        .withColumn("_silver_timestamp", F.current_timestamp()) \
        .drop("_source_file", "data_str")
    
    df_silver.write \
        .format("delta") \
        .mode("overwrite") \
        .option("overwriteSchema", "true") \
        .saveAsTable(f"silver.{tabela_silver}")

# ── GOLD (Business / Serving) ─────────────────────────────
"""
GOLD = Dados prontos para consumo
  • Agregações de negócio
  • Métricas e KPIs
  • Otimizados para consulta (wide tables)
  • Para BI, dashboards, ML
"""

def construir_gold_vendas():
    df_vendas    = spark.read.table("silver.vendas")
    df_clientes  = spark.read.table("silver.clientes")
    df_produtos  = spark.read.table("silver.produtos")
    
    df_gold = df_vendas \
        .join(df_clientes, "cliente_id", "left") \
        .join(df_produtos, "produto_id", "left") \
        .groupBy("ano", "mes", "categoria", "regiao") \
        .agg(
            F.sum("valor_total").alias("receita"),
            F.count("pedido_id").alias("qtd_pedidos"),
            F.countDistinct("cliente_id").alias("clientes_unicos"),
            F.avg("ticket_medio").alias("ticket_medio")
        ) \
        .withColumn("receita_formatada", F.format_number(F.col("receita"), 2))
    
    df_gold.write \
        .format("delta") \
        .mode("overwrite") \
        .option("delta.autoOptimize.optimizeWrite", "true") \
        .option("delta.autoOptimize.autoCompact", "true") \
        .saveAsTable("gold.fato_vendas_mensal")
```

### 6. AutoLoader (Ingestão Incremental)

```python
# ── AutoLoader — detecta novos arquivos automaticamente ────
df_stream = (spark.readStream
    .format("cloudFiles")
    .option("cloudFiles.format", "json")
    .option("cloudFiles.schemaLocation", "/checkpoints/schema/pedidos")
    .option("cloudFiles.inferColumnTypes", "true")
    .option("cloudFiles.schemaEvolutionMode", "addNewColumns")
    .load("abfss://raw@storage.dfs.core.windows.net/pedidos/")
)

# Processar e salvar
df_processado = df_stream \
    .withColumn("_loaded_at", F.current_timestamp()) \
    .withColumn("_file", F.col("_metadata.file_path"))

query = (df_processado.writeStream
    .format("delta")
    .outputMode("append")
    .option("checkpointLocation", "/checkpoints/bronze/pedidos")
    .option("mergeSchema", "true")
    .trigger(availableNow=True)   # Executa uma vez e termina
    .table("bronze.pedidos")
)

query.awaitTermination()
print("✅ AutoLoader concluído!")
```

### 7. Delta Live Tables (DLT)

```python
# ── Pipeline DLT (definido em notebook) ──────────────────
import dlt
from pyspark.sql import functions as F

# Bronze
@dlt.table(
    name="bronze_pedidos",
    comment="Dados brutos de pedidos do e-commerce",
    table_properties={"quality": "bronze"}
)
def bronze_pedidos():
    return (spark.readStream
        .format("cloudFiles")
        .option("cloudFiles.format", "json")
        .load("abfss://raw@storage.dfs.core.windows.net/pedidos/")
    )

# Silver com validação de qualidade
@dlt.table(name="silver_pedidos", comment="Pedidos limpos e validados")
@dlt.expect("id_nao_nulo", "pedido_id IS NOT NULL")
@dlt.expect("valor_positivo", "valor_total > 0")
@dlt.expect_or_drop("data_valida", "data_pedido >= '2020-01-01'")
def silver_pedidos():
    return (dlt.read_stream("bronze_pedidos")
        .dropDuplicates(["pedido_id"])
        .withColumn("valor_total", F.col("valor_total").cast("double"))
        .withColumn("data_pedido", F.to_date("data_pedido"))
    )

# Gold
@dlt.table(name="gold_resumo_diario", comment="KPIs diários de vendas")
def gold_resumo_diario():
    return (dlt.read("silver_pedidos")
        .groupBy("data_pedido", "categoria")
        .agg(
            F.sum("valor_total").alias("receita"),
            F.count("*").alias("total_pedidos"),
            F.avg("valor_total").alias("ticket_medio")
        )
    )
```

### 8. MLflow no Databricks

```python
import mlflow
import mlflow.sklearn
from sklearn.ensemble import GradientBoostingClassifier
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.metrics import accuracy_score, f1_score, roc_auc_score
import pandas as pd
import numpy as np

# ── Configurar MLflow ─────────────────────────────────────
mlflow.set_experiment("/Users/joao@empresa.com/experimento-churn")

# ── Treinar com rastreamento ──────────────────────────────
def treinar_modelo(df_spark, target_col: str = "churn"):
    df = df_spark.toPandas()
    X = df.drop(columns=[target_col])
    y = df[target_col]
    
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.2, stratify=y, random_state=42
    )
    
    params = {
        "n_estimators": [100, 200],
        "max_depth": [3, 5, 7],
        "learning_rate": [0.05, 0.1]
    }
    
    with mlflow.start_run(run_name="GBT_GridSearch"):
        mlflow.set_tags({
            "modelo": "GradientBoosting",
            "dataset": "churn_v2",
            "responsavel": "joao.silva"
        })
        
        gbt = GradientBoostingClassifier(random_state=42)
        grid = GridSearchCV(gbt, params, cv=5, scoring="roc_auc", n_jobs=-1)
        grid.fit(X_train, y_train)
        
        melhor_modelo = grid.best_estimator_
        
        # Log de parâmetros
        mlflow.log_params(grid.best_params_)
        
        # Log de métricas
        y_pred = melhor_modelo.predict(X_test)
        y_proba = melhor_modelo.predict_proba(X_test)[:, 1]
        
        mlflow.log_metrics({
            "accuracy":  accuracy_score(y_test, y_pred),
            "f1_score":  f1_score(y_test, y_pred),
            "roc_auc":   roc_auc_score(y_test, y_proba)
        })
        
        # Registrar modelo
        mlflow.sklearn.log_model(
            melhor_modelo,
            artifact_path="modelo",
            registered_model_name="churn-predictor",
            input_example=X_test.head(5)
        )
        
        run_id = mlflow.active_run().info.run_id
        print(f"✅ Run registrado: {run_id}")
        return melhor_modelo, run_id

# ── Carregar e aplicar modelo ─────────────────────────────
def aplicar_modelo(run_id: str, df_novos):
    model_uri = f"runs:/{run_id}/modelo"
    modelo = mlflow.sklearn.load_model(model_uri)
    
    df_pandas = df_novos.toPandas()
    probabilidades = modelo.predict_proba(df_pandas)[:, 1]
    
    df_pandas["prob_churn"] = probabilidades
    df_pandas["classificacao"] = (probabilidades >= 0.5).astype(int)
    
    return spark.createDataFrame(df_pandas)
```

### 10. Unity Catalog — Governança

```sql
-- ── Hierarquia: Catalog > Schema > Table ──────────────────

-- Criar estrutura
CREATE CATALOG IF NOT EXISTS producao
  COMMENT 'Catálogo de produção';

CREATE SCHEMA IF NOT EXISTS producao.gold
  COMMENT 'Camada Gold — dados prontos para consumo';

-- Criar tabela gerenciada
CREATE TABLE IF NOT EXISTS producao.gold.fato_vendas (
    pedido_id   BIGINT NOT NULL,
    cliente_id  BIGINT,
    data_pedido DATE,
    valor_total DECIMAL(15, 2),
    categoria   STRING
)
USING DELTA
COMMENT 'Fato de vendas consolidado'
TBLPROPERTIES (
    'delta.enableChangeDataFeed' = 'true',
    'delta.autoOptimize.optimizeWrite' = 'true'
);

-- ── Controle de acesso ────────────────────────────────────
GRANT USE CATALOG  ON CATALOG producao TO `grupo-dados`;
GRANT USE SCHEMA   ON SCHEMA producao.gold TO `grupo-dados`;
GRANT SELECT       ON TABLE producao.gold.fato_vendas TO `analistas`;
GRANT MODIFY       ON TABLE producao.gold.fato_vendas TO `engenheiros`;

-- ── Row-Level Security com Row Filters ───────────────────
CREATE FUNCTION producao.gold.filtro_regiao(regiao STRING)
  RETURNS BOOLEAN
  RETURN IS_ACCOUNT_GROUP_MEMBER(regiao) OR IS_ACCOUNT_GROUP_MEMBER('admin');

ALTER TABLE producao.gold.fato_vendas
  SET ROW FILTER producao.gold.filtro_regiao ON (regiao);

-- ── Lineage ───────────────────────────────────────────────
-- Automático no Unity Catalog — visualize no UI:
-- Data Explorer → Tabela → Lineage tab
```
